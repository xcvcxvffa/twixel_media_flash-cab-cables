Options -Indexes

RewriteEngine On
RewriteBase /

# ─── PHP API routing ──────────────────────────────────────────────────────────
# All /api/* requests → api/index.php (PHP backend)
RewriteCond %{REQUEST_URI} ^/api/ [NC]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteRule ^api/(.*)$ api/index.php [QSA,L]

# ─── Admin SPA routing ────────────────────────────────────────────────────────
# All /admin/* requests → admin/index.html (React SPA fallback)
RewriteCond %{REQUEST_URI} ^/admin/ [NC]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^admin/(.*)$ /admin/index.html [L]

# ─── Client SPA routing ───────────────────────────────────────────────────────
# Serve existing files/dirs normally
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# For all other routes → index.html (React client SPA)
RewriteCond %{REQUEST_URI} !^/api/ [NC]
RewriteCond %{REQUEST_URI} !^/admin/ [NC]
RewriteRule ^ /index.html [L]

# ─── Cache headers for static assets ─────────────────────────────────────────
<FilesMatch "\.(js|css|woff2|woff|ttf|svg|png|jpg|jpeg|webp|gif|ico)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
</FilesMatch>

<FilesMatch "\.(html)$">
    Header set Cache-Control "no-cache, no-store, must-revalidate"
</FilesMatch>

# PHP upload limits (for admin image uploads)
php_value upload_max_filesize 50M
php_value post_max_size 50M
php_value memory_limit 256M
